% --- Safe Package Loading for Pandoc ---
\PassOptionsToPackage{svgnames}{xcolor}
\PassOptionsToPackage{most}{tcolorbox}

\usepackage{tcolorbox}
\usepackage{environ}
\usepackage{keyval}

% Pandoc's default template loads xcolor (without options) before
% header-includes run, so PassOptionsToPackage above arrives too late
% to enable svgnames. Load the SVG color names directly as a fallback;
% re-running this after a successful svgnames load is harmless.
\input{svgnam.def}

% --- Define the Keys ---
\makeatletter
\define@key{pullquote}{width}{\def\pqwidth{#1}}
\define@key{pullquote}{color}{\def\pqcolor{#1}}
\define@key{pullquote}{size}{\def\pqsize{#1}}
\define@key{pullquote}{skip}{\def\pqskip{#1}}
\define@key{pullquote}{align}{\def\pqalign{#1}}
\define@key{pullquote}{boxalign}{\def\pqboxalign{#1}}
\define@key{pullquote}{barwidth}{\def\pqbarwidth{#1}}
\define@key{pullquote}{barcolor}{\def\pqbarcolor{#1}}
\define@key{pullquote}{paddingleft}{\def\pqpaddingleft{#1}}
\define@key{pullquote}{paddingright}{\def\pqpaddingright{#1}}
\define@key{pullquote}{paddingtop}{\def\pqpaddingtop{#1}}
\define@key{pullquote}{paddingbottom}{\def\pqpaddingbottom{#1}}
\makeatother

% --- Publicly Sensible Default Values ---
\newcommand{\pqsetdefaults}{%
  \def\pqwidth{0.8\textwidth}%
  \def\pqcolor{DarkGray}%
  \def\pqsize{\large\itshape}%
  \def\pqskip{1.2\baselineskip}%
  \def\pqalign{\raggedright}%
  \def\pqboxalign{center}%
  \def\pqbarwidth{4pt}%
  \def\pqbarcolor{LightGray}%
  \def\pqpaddingleft{12pt}%
  \def\pqpaddingright{0pt}%
  \def\pqpaddingtop{4pt}%
  \def\pqpaddingbottom{4pt}%
}

% --- Environment Definition ---
\NewEnviron{pullquote}[1][]{%
  \pqsetdefaults
  \setkeys{pullquote}{#1}%
  %
  \begin{tcolorbox}[
    enhanced,       % Enables advanced border controls
    sharp corners,
    colback=white,
    frame hidden,   % Completely disables the base bounding box (kills the hairline)
    borderline west={\pqbarwidth}{0pt}{\pqbarcolor}, % Explicitly draws ONLY the left bar
    width=\pqwidth,
    \pqboxalign,
    left=\pqpaddingleft, right=\pqpaddingright, top=\pqpaddingtop, bottom=\pqpaddingbottom,
    before skip=15pt, after skip=15pt,
    colupper=\pqcolor,
    before upper={\pqsize\setlength{\baselineskip}{\pqskip}\pqalign},
  ]
   \BODY
  \end{tcolorbox}
}
