% --- Translation Layer ---
\newcommand{\pqapplymappings}{%
  % Semantic Size mapping (Calculated dynamically relative to current font size)
  \edef\pqcurrentsize{\f@size}%
  \def\pqsize{\pqrawsize}%
  \ifdefstring{\pqrawsize}{s}{\def\pqsize{\fontsize{\fpeval{0.85 * \pqcurrentsize}pt}{\fpeval{1.02 * \pqcurrentsize}pt}\selectfont}}{}%
  \ifdefstring{\pqrawsize}{m}{\def\pqsize{\fontsize{\fpeval{1.0 * \pqcurrentsize}pt}{\fpeval{1.2 * \pqcurrentsize}pt}\selectfont}}{}%
  \ifdefstring{\pqrawsize}{l}{\def\pqsize{\fontsize{\fpeval{1.2 * \pqcurrentsize}pt}{\fpeval{1.44 * \pqcurrentsize}pt}\selectfont}}{}%
  \ifdefstring{\pqrawsize}{xl}{\def\pqsize{\fontsize{\fpeval{1.5 * \pqcurrentsize}pt}{\fpeval{1.8 * \pqcurrentsize}pt}\selectfont}}{}%
  \ifdefstring{\pqrawsize}{2xl}{\def\pqsize{\fontsize{\fpeval{2.0 * \pqcurrentsize}pt}{\fpeval{2.4 * \pqcurrentsize}pt}\selectfont}}{}%

  \def\pqboxalign{center}% structure default
  \ifdefstring{\pqrawboxalign}{left}{\def\pqboxalign{flush left}}{}%
  \ifdefstring{\pqrawboxalign}{right}{\def\pqboxalign{flush right}}{}%

  \def\pqtextalign{}% default to inherit
  \ifdefstring{\pqrawtextalign}{center}{\def\pqtextalign{\centering}}{}%
  \ifdefstring{\pqrawtextalign}{right}{\def\pqtextalign{\raggedleft}}{}%
  \ifdefstring{\pqrawtextalign}{justify}{\def\pqtextalign{\relax}}{}%

  \def\pqweight{}% default to inherit
  \ifdefstring{\pqrawweight}{normal}{\def\pqweight{\mdseries}}{}%
  \ifdefstring{\pqrawweight}{bold}{\def\pqweight{\bfseries}}{}%
  \ifdefstring{\pqrawweight}{medium}{\def\pqweight{\bfseries}}{}%

  \def\pqstyle{}% default to inherit
  \ifdefstring{\pqrawstyle}{italic}{\def\pqstyle{\itshape}}{}%
  \ifdefstring{\pqrawstyle}{upright}{\def\pqstyle{\upshape}}{}%
  \ifdefstring{\pqrawstyle}{slanted}{\def\pqstyle{\slshape}}{}%
  \ifdefstring{\pqrawstyle}{smallcaps}{\def\pqstyle{\scshape}}{}%

  \def\pqfont{}% default to inherit
  \ifdefempty{\pqrawfont}{}{%
    \def\pqfont{\fontspec{\pqrawfont}}% literal override
  }%

  % Semantic Line Spacing (Math calculated against baseline)
  \def\pqskip{}% default
  \ifdefempty{\pqrawskip}{}{%
    \def\pqskip{\linespread{\fpeval{\pqrawskip / 1.2}}}%
  }%
}
\makeatother

% --- Empty Typography Defaults ---
\newcommand{\pqsetdefaults}{%
  \def\pqwidth{0.8\linewidth}%
  \def\pqcolor{}% Empty inherits current \color
  \def\pqbarwidth{4pt}%
  \def\pqbarcolor{pqdefaultbar}%
  \def\pqpaddingleft{12pt}%
  \def\pqpaddingright{0pt}%
  \def\pqpaddingtop{4pt}%
  \def\pqpaddingbottom{4pt}%
  \def\pqrawsize{}%
  \def\pqrawtextalign{}%
  \def\pqrawboxalign{center}%
  \def\pqrawweight{}%
  \def\pqrawstyle{}%
  \def\pqrawfont{}%
  \def\pqrawskip{}%
}

% --- Environment Definition ---
\newenvironment{pullquote}[1][]{%
  \pqsetdefaults
  \setkeys{pullquote}{#1}%
  \pqapplymappings
  %
  \begin{tcolorbox}[
    enhanced,
    sharp corners,
    colback=white,
    frame hidden,
    borderline west={\pqbarwidth}{0pt}{\pqbarcolor},
    width=\pqwidth,
    \pqboxalign,
    left=\pqpaddingleft, right=\pqpaddingright, top=\pqpaddingtop, bottom=\pqpaddingbottom,
    before skip=15pt, after skip=15pt,
    % Apply typography stack sequentially
    before upper={%
      \ifdefempty{\pqcolor}{}{\color{\pqcolor}}%
      \pqsize\pqfont\pqweight\pqstyle
      \pqskip\selectfont
      \setlength{\parskip}{0.5\baselineskip}%
      \ifdefempty{\pqtextalign}{}{\pqtextalign}%
    },
  ]
}{%
  \end{tcolorbox}
}
\makeatother
